# EdgeDesk — what crawlers may read.
#
# The public surface is the landing page, the RESEARCH ARTICLES under
# /articles/, EdgeDesk Games and the legal pages. The research terminal and
# the Collective are application shells behind an account: they render nothing
# useful to a crawler and are not offered.
#
# Articles are static HTML with no account in front of them: everything a
# crawler needs is in the markup before any script runs. Only PUBLISHED
# articles exist as files at all — a draft is never written to /articles/, and
# the one place an unpublished one can be rendered, /articles/_preview/, is
# disallowed here, marked noindex in its own head and absent from every
# sitemap.
User-agent: *
Allow: /
Allow: /articles
Allow: /articles/

# Member posts and the composer are NOT disallowed, on purpose. Both carry
# `noindex` in their own head, and a crawler that is blocked from fetching a
# page cannot read the noindex on it — a disallowed URL can still end up in the
# index, listed with no content at all. Allowing the crawl is what makes the
# noindex land. They appear in no sitemap either way, and member posts are
# written by members rather than by EdgeDesk, which is why they are kept out of
# search while the research articles are put in it.
Allow: /games
Allow: /games/

# THE FREE TOOLS live under /tools/, which is ALSO where the repository keeps
# its build and test scripts. Only the tool pages and their stylesheet are
# offered; everything else under /tools/ stays out. Google and Bing apply the
# longest matching rule, so each Allow below wins over the Disallow for its
# own path and nothing else. ($ = this exact URL.)
Allow: /tools/$
Allow: /tools/tools.css
Allow: /tools/no-vig-calculator/
Allow: /tools/fair-odds-calculator/
Allow: /tools/model-vs-market/
Disallow: /tools/

# The article store: every record, published or not, as JSON. Drafts are not
# for search; the published index the explorer reads is.
Allow: /articles/data/published.json
Disallow: /articles/data/
Disallow: /supabase/
Disallow: /articles/_preview/
Disallow: /admin/
Disallow: /admin.html
Disallow: /collective/admin.html
Disallow: /reset.html
Disallow: /join/
